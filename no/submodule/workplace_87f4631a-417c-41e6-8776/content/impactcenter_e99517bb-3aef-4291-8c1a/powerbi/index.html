    Map To Choices

    Create a New Custom Column in the Semantic Model
    let
    selectedValues =
        if [icci_program] = null then
            {}
        else
            List.Transform(
                Text.Split(Text.From([icci_program]), ","),
                each Text.Trim(_)
            ),

    programChoices =
        Table.SelectRows(
            Choices,
            each [icci_group] = "YOUR CHOICE GROUP NAME"
        ),

    labels =
        List.Transform(
            selectedValues,
            (value) =>
                let
                    match =
                        Table.SelectRows(
                            programChoices,
                            each Text.From([icci_value]) = value
                        )
                in
                    if Table.IsEmpty(match) then
                        value
                    else
                        match{0}[icci_code]
        )
in
    Text.Combine(labels, ", ")
