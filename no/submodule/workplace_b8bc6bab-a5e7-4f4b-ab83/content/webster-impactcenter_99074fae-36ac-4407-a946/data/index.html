    $TOKEN = az account get-access-token `
  --resource "https://orgc302603f.crm.dynamics.com" `
  --query accessToken `
  --output tsv

$headers = @{
    Authorization = "Bearer $TOKEN"
    Accept        = "application/json"
}

$url = "https://orgc302603f.crm.dynamics.com/api/data/v9.2/EntityDefinitions(LogicalName='cr34c_pageevent')?`$expand=ManyToOneRelationships"
$url = "https://orgc302603f.crm.dynamics.com/api/data/v9.2/EntityDefinitions(LogicalName='cr34c_pageevent')?`$select=LogicalName,SchemaName&`$expand=ManyToOneRelationships(`$select=SchemaName,ReferencedEntity,ReferencingEntity,ReferencingAttribute,ReferencedAttribute)"

try {
    $response = Invoke-RestMethod `
        -Uri $url `
        -Headers $headers `
        -Method Get `
        -ErrorAction Stop

    $response | ConvertTo-Json -Depth 20
}
catch {
    Write-Host "===== ERROR =====" -ForegroundColor Red
    Write-Host $_.Exception.Message

    Write-Host ""
    Write-Host "===== EXCEPTION TYPE =====" -ForegroundColor Yellow
    Write-Host $_.Exception.GetType().FullName

    Write-Host ""
    Write-Host "===== HTTP STATUS =====" -ForegroundColor Yellow
    if ($_.Exception.Response) {
        Write-Host $_.Exception.Response.StatusCode
        Write-Host $_.Exception.Response.StatusDescription
    }

    Write-Host ""
    Write-Host "===== RESPONSE BODY =====" -ForegroundColor Yellow

    if ($_.ErrorDetails.Message) {
        Write-Host $_.ErrorDetails.Message
    }

    Write-Host ""
    Write-Host "===== FULL ERROR =====" -ForegroundColor Yellow
    $_ | Format-List * -Force
}
